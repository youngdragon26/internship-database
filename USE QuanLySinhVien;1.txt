USE QuanLySinhVien;

-- 1. Sinh viên có tên bắt đầu bằng ký tự 'h'
SELECT *
FROM Student
WHERE StudentName LIKE 'h%';

-- 2. Lớp học có thời gian bắt đầu vào tháng 12
SELECT *
FROM Class
WHERE MONTH(StartDate) = 12;

-- 3. Môn học có credit trong khoảng từ 3 đến 5
SELECT *
FROM Subject
WHERE Credit BETWEEN 3 AND 5;

-- 4. Đổi mã lớp của sinh viên tên 'Hung' thành 2
SET SQL_SAFE_UPDATES = 0;
UPDATE Student
SET ClassId = 2
WHERE StudentName = 'Hung';
SET SQL_SAFE_UPDATES = 1;

-- 5. StudentName, SubName, Mark; điểm giảm dần, trùng thì tên tăng dần
SELECT S.StudentName, Sub.SubName, M.Mark
FROM Student S
JOIN Mark M ON S.StudentId = M.StudentId
JOIN Subject Sub ON M.SubId = Sub.SubId
ORDER BY M.Mark DESC, S.StudentName ASC;
