-- PHẦN 1: Tạo cơ sở dữ liệu (bỏ qua nếu bạn đã có từ bài trước)
CREATE DATABASE IF NOT EXISTS QuanLyBanHang;
USE QuanLyBanHang;

CREATE TABLE IF NOT EXISTS Customer(
    cID INT PRIMARY KEY,
    cName VARCHAR(25),
    cAge TINYINT
);

CREATE TABLE IF NOT EXISTS `Order`(
    oID INT PRIMARY KEY,
    cID INT,
    oDate DATETIME,
    oTotalPrice INT,
    FOREIGN KEY (cID) REFERENCES Customer(cID)
);

CREATE TABLE IF NOT EXISTS Product(
    pID INT PRIMARY KEY,
    pName VARCHAR(25),
    pPrice INT
);

CREATE TABLE IF NOT EXISTS OrderDetail(
    oID INT,
    pID INT,
    odQTY INT,
    PRIMARY KEY (oID, pID),
    FOREIGN KEY (oID) REFERENCES `Order`(oID),
    FOREIGN KEY (pID) REFERENCES Product(pID)
);

-- PHẦN 3: Truy vấn

-- 1. oID, oDate, oPrice của tất cả các hóa đơn
SELECT oID, oDate, oTotalPrice AS oPrice
FROM `Order`;

-- 2. Khách hàng đã mua hàng và các sản phẩm họ đã mua
SELECT DISTINCT C.cName, P.pName
FROM Customer C
JOIN `Order` O ON C.cID = O.cID
JOIN OrderDetail OD ON O.oID = OD.oID
JOIN Product P ON OD.pID = P.pID;

-- 3. Tên khách hàng không mua sản phẩm nào
SELECT C.cName
FROM Customer C
LEFT JOIN `Order` O ON C.cID = O.cID
WHERE O.oID IS NULL;

-- 4. Mã hóa đơn, ngày bán và giá tiền từng hóa đơn
SELECT O.oID, O.oDate, SUM(OD.odQTY * P.pPrice) AS oPrice
FROM `Order` O
JOIN OrderDetail OD ON O.oID = OD.oID
JOIN Product P ON OD.pID = P.pID
GROUP BY O.oID, O.oDate;
